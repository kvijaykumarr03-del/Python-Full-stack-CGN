# Hard SQL Queries — Vehicle Rental System

USE vehicle_rental_db;

-- Q1. Find the vehicle model that generated the highest revenue from completed rentals
SELECT v.model_name, v.brand, SUM(r.total_amount) AS total_revenue
FROM vehicles v
JOIN rentals r ON v.vehicle_id = r.vehicle_id
WHERE r.rental_status = 'Completed'
GROUP BY v.vehicle_id, v.model_name, v.brand
ORDER BY total_revenue DESC
LIMIT 1;

-- Q2. Find the customer who spent the most on completed rentals
SELECT c.customer_id, c.customer_name, c.city, SUM(r.total_amount) AS total_spent
FROM customers c
JOIN rentals r ON c.customer_id = r.customer_id
WHERE r.rental_status = 'Completed'
GROUP BY c.customer_id, c.customer_name, c.city
ORDER BY total_spent DESC
LIMIT 1;

-- Q3. Find the pickup location generating the highest revenue from completed rentals
SELECT loc.location_id, loc.location_name, loc.city, SUM(r.total_amount) AS total_revenue
FROM rental_locations loc
JOIN rentals r ON loc.location_id = r.pickup_location_id
WHERE r.rental_status = 'Completed'
GROUP BY loc.location_id, loc.location_name, loc.city
ORDER BY total_revenue DESC
LIMIT 1;

-- Q4. Find the second-highest individual rental transaction amount
SELECT MAX(total_amount) AS second_highest_amount
FROM rentals
WHERE total_amount < (
    SELECT MAX(total_amount) FROM rentals
);

-- Q5. Find customers whose completed rental spending is higher than the average customer spending
WITH customer_spending AS (
    SELECT customer_id, SUM(total_amount) AS total_spent
    FROM rentals
    WHERE rental_status = 'Completed'
    GROUP BY customer_id
)
SELECT c.customer_id, c.customer_name, c.city, cs.total_spent
FROM customer_spending cs
JOIN customers c ON cs.customer_id = c.customer_id
WHERE cs.total_spent > (
    SELECT AVG(total_spent) FROM customer_spending
)
ORDER BY cs.total_spent DESC;

-- Q6. Rank all vehicles by total completed revenue using DENSE_RANK()
SELECT v.model_name, v.brand, v.category,
       SUM(r.total_amount) AS total_revenue,
       DENSE_RANK() OVER (
           ORDER BY SUM(r.total_amount) DESC
       ) AS revenue_rank
FROM vehicles v
JOIN rentals r ON v.vehicle_id = r.vehicle_id
WHERE r.rental_status = 'Completed'
GROUP BY v.vehicle_id, v.model_name, v.brand, v.category;

-- Q7. Rank customers within their respective city by total spending on completed rentals
WITH city_spending AS (
    SELECT c.customer_id, c.customer_name, c.city,
           SUM(r.total_amount) AS total_spent
    FROM customers c
    JOIN rentals r ON c.customer_id = r.customer_id
    WHERE r.rental_status = 'Completed'
    GROUP BY c.customer_id, c.customer_name, c.city
)
SELECT customer_name, city, total_spent,
       DENSE_RANK() OVER (
           PARTITION BY city
           ORDER BY total_spent DESC
       ) AS city_spending_rank
FROM city_spending;

-- Q8. Find the top 3 highest-earning vehicle models in each vehicle category
WITH category_revenue AS (
    SELECT v.category, v.model_name, v.brand,
           SUM(r.total_amount) AS total_revenue
    FROM vehicles v
    JOIN rentals r ON v.vehicle_id = r.vehicle_id
    WHERE r.rental_status = 'Completed'
    GROUP BY v.category, v.model_name, v.brand
),
ranked_category AS (
    SELECT *,
           ROW_NUMBER() OVER (
               PARTITION BY category
               ORDER BY total_revenue DESC
           ) AS rn
    FROM category_revenue
)
SELECT category, model_name, brand, total_revenue
FROM ranked_category
WHERE rn <= 3
ORDER BY category, total_revenue DESC;

-- Q9. Find the top 3 spending customers overall from completed rentals
SELECT c.customer_id, c.customer_name, c.email, c.city,
       SUM(r.total_amount) AS total_spent
FROM customers c
JOIN rentals r ON c.customer_id = r.customer_id
WHERE r.rental_status = 'Completed'
GROUP BY c.customer_id, c.customer_name, c.email, c.city
ORDER BY total_spent DESC
LIMIT 3;

-- Q10. Find the most-rented vehicle model by count of rental bookings
SELECT v.model_name, v.brand, COUNT(r.rental_id) AS rental_count
FROM vehicles v
JOIN rentals r ON v.vehicle_id = r.vehicle_id
GROUP BY v.vehicle_id, v.model_name, v.brand
ORDER BY rental_count DESC
LIMIT 1;

-- Q11. Find the customer who has completed the highest number of rentals
SELECT c.customer_id, c.customer_name, COUNT(r.rental_id) AS completed_rentals_count
FROM customers c
JOIN rentals r ON c.customer_id = r.customer_id
WHERE r.rental_status = 'Completed'
GROUP BY c.customer_id, c.customer_name
ORDER BY completed_rentals_count DESC
LIMIT 1;

-- Q12. Find the vehicle brand that has the highest average rental transaction amount
SELECT v.brand, ROUND(AVG(r.total_amount), 2) AS avg_transaction_amount
FROM vehicles v
JOIN rentals r ON v.vehicle_id = r.vehicle_id
GROUP BY v.brand
ORDER BY avg_transaction_amount DESC
LIMIT 1;

-- Q13. Calculate each vehicle category's percentage contribution to completed rental revenue
SELECT v.category,
       SUM(r.total_amount) AS category_revenue,
       ROUND(
           SUM(r.total_amount) * 100.0 /
           (SELECT SUM(total_amount) FROM rentals WHERE rental_status = 'Completed'),
           2
       ) AS revenue_percentage
FROM vehicles v
JOIN rentals r ON v.vehicle_id = r.vehicle_id
WHERE r.rental_status = 'Completed'
GROUP BY v.category
ORDER BY category_revenue DESC;

-- Q14. Find the busiest pickup location based on total rental days generated
SELECT loc.location_name, loc.city, SUM(r.rental_duration_days) AS total_rental_days
FROM rental_locations loc
JOIN rentals r ON loc.location_id = r.pickup_location_id
GROUP BY loc.location_id, loc.location_name, loc.city
ORDER BY total_rental_days DESC
LIMIT 1;

-- Q15. Find customers who booked vehicles for more total days than the overall customer average
SELECT c.customer_id, c.customer_name, SUM(r.rental_duration_days) AS total_days_rented
FROM customers c
JOIN rentals r ON c.customer_id = r.customer_id
GROUP BY c.customer_id, c.customer_name
HAVING SUM(r.rental_duration_days) > (
    SELECT AVG(customer_days)
    FROM (
        SELECT SUM(rental_duration_days) AS customer_days
        FROM rentals
        GROUP BY customer_id
    ) AS avg_table
)
ORDER BY total_days_rented DESC;

-- Q16. Find the highest-revenue vehicle model for each brand from completed rentals
WITH brand_models AS (
    SELECT v.brand, v.model_name,
           SUM(r.total_amount) AS model_revenue
    FROM vehicles v
    JOIN rentals r ON v.vehicle_id = r.vehicle_id
    WHERE r.rental_status = 'Completed'
    GROUP BY v.brand, v.model_name
),
ranked_models AS (
    SELECT *,
           ROW_NUMBER() OVER (
               PARTITION BY brand
               ORDER BY model_revenue DESC
           ) AS rn
    FROM brand_models
)
SELECT brand, model_name, model_revenue
FROM ranked_models
WHERE rn = 1
ORDER BY model_revenue DESC;

-- Q17. Find repeat customers who have completed 4 or more rentals
SELECT c.customer_id, c.customer_name, c.city, COUNT(r.rental_id) AS rental_count,
       SUM(r.total_amount) AS total_spent
FROM customers c
JOIN rentals r ON c.customer_id = r.customer_id
WHERE r.rental_status = 'Completed'
GROUP BY c.customer_id, c.customer_name, c.city
HAVING COUNT(r.rental_id) >= 4
ORDER BY rental_count DESC;

-- Q18. Find the highest-value rental transaction for each payment method
WITH ranked_payments AS (
    SELECT payment_method, rental_id, customer_id, total_amount,
           ROW_NUMBER() OVER (
               PARTITION BY payment_method
               ORDER BY total_amount DESC
           ) AS rn
    FROM rentals
)
SELECT payment_method, rental_id, customer_id, total_amount
FROM ranked_payments
WHERE rn = 1;

-- Q19. Compare revenue and count between Completed and Cancelled rentals
SELECT rental_status,
       COUNT(*) AS total_count,
       SUM(total_amount) AS total_amount_sum,
       ROUND(AVG(total_amount), 2) AS avg_amount
FROM rentals
WHERE rental_status IN ('Completed', 'Cancelled')
GROUP BY rental_status;

-- Q20. Retrieve each customer's most recent rental using a window function
WITH customer_latest AS (
    SELECT r.customer_id, r.rental_id, r.rental_start_date, r.total_amount, r.rental_status,
           ROW_NUMBER() OVER (
               PARTITION BY r.customer_id
               ORDER BY r.rental_start_date DESC
           ) AS rn
    FROM rentals r
)
SELECT c.customer_name, cl.rental_id, cl.rental_start_date, cl.total_amount, cl.rental_status
FROM customer_latest cl
JOIN customers c ON cl.customer_id = c.customer_id
WHERE cl.rn = 1;
