# Medium SQL Queries — Vehicle Rental System

USE vehicle_rental_db;

-- Q1. Display rentals along with customer names and cities
SELECT r.rental_id, c.customer_name, c.city, r.rental_start_date, r.total_amount
FROM rentals r
JOIN customers c ON r.customer_id = c.customer_id;

-- Q2. Display rentals with vehicle model, brand, and category
SELECT r.rental_id, v.model_name, v.brand, v.category, r.rental_duration_days, r.total_amount
FROM rentals r
JOIN vehicles v ON r.vehicle_id = v.vehicle_id;

-- Q3. Display rentals with pickup location name and city
SELECT r.rental_id, loc.location_name AS pickup_location, loc.city, r.rental_start_date
FROM rentals r
JOIN rental_locations loc ON r.pickup_location_id = loc.location_id;

-- Q4. Display complete rental details using all four tables
SELECT r.rental_id, c.customer_name, v.model_name, v.brand,
       pl.location_name AS pickup_location, rl.location_name AS return_location,
       r.rental_start_date, r.rental_duration_days, r.total_amount, r.rental_status
FROM rentals r
JOIN customers c ON r.customer_id = c.customer_id
JOIN vehicles v ON r.vehicle_id = v.vehicle_id
JOIN rental_locations pl ON r.pickup_location_id = pl.location_id
JOIN rental_locations rl ON r.return_location_id = rl.location_id;

-- Q5. Count total rentals made by each customer
SELECT c.customer_id, c.customer_name, COUNT(r.rental_id) AS total_rentals
FROM customers c
LEFT JOIN rentals r ON c.customer_id = r.customer_id
GROUP BY c.customer_id, c.customer_name;

-- Q6. Calculate total revenue generated by each vehicle from completed rentals
SELECT v.vehicle_id, v.brand, v.model_name, SUM(r.total_amount) AS total_revenue
FROM vehicles v
JOIN rentals r ON v.vehicle_id = r.vehicle_id
WHERE r.rental_status = 'Completed'
GROUP BY v.vehicle_id, v.brand, v.model_name;

-- Q7. Calculate total rentals and revenue by vehicle category
SELECT v.category, COUNT(r.rental_id) AS total_rentals, SUM(r.total_amount) AS total_revenue
FROM vehicles v
JOIN rentals r ON v.vehicle_id = r.vehicle_id
WHERE r.rental_status = 'Completed'
GROUP BY v.category;

-- Q8. Find average rental amount by payment method
SELECT payment_method, ROUND(AVG(total_amount), 2) AS average_rental_amount
FROM rentals
GROUP BY payment_method;

-- Q9. Find pickup locations that have handled more than 150 rentals
SELECT loc.location_id, loc.location_name, loc.city, COUNT(r.rental_id) AS total_pickups
FROM rental_locations loc
JOIN rentals r ON loc.location_id = r.pickup_location_id
GROUP BY loc.location_id, loc.location_name, loc.city
HAVING COUNT(r.rental_id) > 150;

-- Q10. Find vehicle brands with total completed-rental revenue greater than 500,000
SELECT v.brand, SUM(r.total_amount) AS total_revenue
FROM vehicles v
JOIN rentals r ON v.vehicle_id = r.vehicle_id
WHERE r.rental_status = 'Completed'
GROUP BY v.brand
HAVING SUM(r.total_amount) > 500000;

-- Q11. Find the highest and lowest daily rates across all vehicle categories
SELECT category, MAX(daily_rate) AS max_daily_rate, MIN(daily_rate) AS min_daily_rate
FROM vehicles
GROUP BY category;

-- Q12. Find the average rental duration for each vehicle category
SELECT v.category, ROUND(AVG(r.rental_duration_days), 2) AS avg_duration_days
FROM vehicles v
JOIN rentals r ON v.vehicle_id = r.vehicle_id
GROUP BY v.category;

-- Q13. Categorize rentals into Budget, Standard, and Premium tiers
SELECT rental_id, total_amount,
       CASE
           WHEN total_amount >= 25000 THEN 'Premium'
           WHEN total_amount >= 10000 THEN 'Standard'
           ELSE 'Budget'
       END AS rental_tier
FROM rentals;

-- Q14. Display rentals scheduled in the year 2025
SELECT * FROM rentals
WHERE YEAR(rental_start_date) = 2025;

-- Q15. Display rentals that started in the month of June
SELECT * FROM rentals
WHERE MONTH(rental_start_date) = 6;

-- Q16. Find customers who have made at least one booking
SELECT customer_id, customer_name, email, city
FROM customers
WHERE customer_id IN (
    SELECT DISTINCT customer_id FROM rentals
);

-- Q17. Find vehicles that have never been rented
SELECT v.vehicle_id, v.model_name, v.brand, v.category
FROM vehicles v
LEFT JOIN rentals r ON v.vehicle_id = r.vehicle_id
WHERE r.rental_id IS NULL;

-- Q18. Find rentals where total amount exceeds the overall average rental amount
SELECT rental_id, customer_id, vehicle_id, total_amount
FROM rentals
WHERE total_amount > (
    SELECT AVG(total_amount) FROM rentals
);

-- Q19. Find the count and total revenue of rentals grouped by rental status
SELECT rental_status,
       COUNT(*) AS total_rentals,
       SUM(total_amount) AS total_amount_sum
FROM rentals
GROUP BY rental_status;

-- Q20. Find total revenue from completed rentals grouped by pickup city
SELECT loc.city, SUM(r.total_amount) AS total_revenue
FROM rental_locations loc
JOIN rentals r ON loc.location_id = r.pickup_location_id
WHERE r.rental_status = 'Completed'
GROUP BY loc.city
ORDER BY total_revenue DESC;
